cmake_minimum_required(VERSION 3.10)
project(VectorStudentGrades LANGUAGES CXX)
add_executable(student_grades main.cpp StudentManager.cpp)
set_target_properties(student_grades PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
    target_compile_options(student_grades PRIVATE /W4 /utf-8)
else()
    target_compile_options(student_grades PRIVATE -Wall -Wextra -pedantic)
endif()
